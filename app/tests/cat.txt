# The Bestiary's cat (FMB Vocal's own, app/dsp/cat.h): LFO opens the Bestiary on the robot (0.4.0: FMB Vocal's
# own first), ALGORITHM two on the cat (a small airway and nose on the waveguide at the full rate). A key held meows at its pitch (C4 an octave up: the
# octave that centres its 120..1600 Hz on middle C); PLAY its meow; PRESET the purr (pulsed, through the nose) and the
# hiss (no voice, air at the teeth); SEL its purr page; kept through a reboot; LFO back to the human voice.
#   python base/tools/build.py app --host; app/build/host/fmb-vocal_host app/tests/cat.txt app/build/host
wait 400
listen
tap LFO
said BESTIARY
wait 300
spin ALGO 2
said CREATURE
wait 600
expect creature 0
expect creaudio 0
wav cat-key.wav
peakreset
key 7 down
wait 900
expect sealgate >90
expect sealhz >515
expect sealhz <530
shot cat.png
key 7 up
wait 400
expect sealgate <5
wavstop
expect peak_db_max -1
expect peak_db_min -30
# PLAY: the meow (0.9 s)
wav cat-meow.wav
peakreset
tap PLAY
wait 200
expect sealcall 0
wait 900
expect sealcall -1
wavstop
expect peak_db_max -1
expect peak_db_min -30
# PRESET: the purr (3.8 s), low and pulsed
spin PRESET 1
expect sealpick 1
said PURR
wav cat-purr.wav
peakreset
tap PLAY
wait 300
expect sealcall 1
shot cat-purr.png
wait 3700
expect sealcall -1
wavstop
expect peak_db_max -1
expect peak_db_min -30
# PRESET: the hiss (1.3 s), no voice
spin PRESET 1
expect sealpick 2
said HISS
wav cat-hiss.wav
peakreset
tap PLAY
wait 300
expect sealcall 2
wait 1200
expect sealcall -1
wavstop
expect peak_db_max -1
expect peak_db_min -45
# SEL: the purr page; kept through a reboot
tap SEL
expect sealpage 1
shot cat-page2.png
tap SEL
tap SAVE
wait 100
said SAVED
reboot
wait 400
tap LFO
wait 400
expect creature 0
expect creaudio 0
expect sealpick 2
# ALGORITHM on: the songbird, next; LFO: the human voice
spin ALGO 1
said SONGBIRD
wait 300
expect creature 1
tap LFO
said HUMAN
wait 300
expect body 0
